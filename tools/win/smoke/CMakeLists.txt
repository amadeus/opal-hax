# Compatibility check: our existing camera tool (tools/opaltool.cpp) built against the self-built depthai-core
# (public source + Opal's firmware, tools/win/build-depthai.sh) instead of Opal.app's private library.
cmake_minimum_required(VERSION 3.16)
project(opal_smoke CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(depthai CONFIG REQUIRED)
add_executable(opaltool-dai ../../opaltool.cpp)
target_link_libraries(opaltool-dai PRIVATE depthai::core)
add_executable(frames frames.cpp)
target_link_libraries(frames PRIVATE depthai::core)
add_executable(verify verify.cpp)
target_link_libraries(verify PRIVATE depthai::core)
add_executable(rawpkt rawpkt.cpp ../core/opal_pipeline.cpp)
target_link_libraries(rawpkt PRIVATE depthai::core)
add_executable(opalstream opalstream.cpp ../core/opal_pipeline.cpp)
target_link_libraries(opalstream PRIVATE depthai::core)
add_executable(decodectl decodectl.cpp)
target_link_libraries(decodectl PRIVATE depthai::core)
add_executable(camtest camtest.cpp ../core/opal_camera.cpp ../core/opal_pipeline.cpp)
target_link_libraries(camtest PRIVATE depthai::core)
add_executable(profile ../../profile/profile.cpp ../core/opal_pipeline.cpp)
target_link_libraries(profile PRIVATE depthai::core)
